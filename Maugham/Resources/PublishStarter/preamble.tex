% --- Bilingual / multi-language bodies ----------------------------------------
% Maugham's build/body.tex wraps EACH language's body in \begin{MaughamBody}{<tag>}
% ... \end{MaughamBody}, in order, and passes the language tag as #1. The default
% below just starts each half on a fresh page. To give each half its own title
% page, move frontmatter's \input inside the environment and key it on #1 — e.g.
%   \renewenvironment{MaughamBody}[1]{\clearpage\input{frontmatter-#1}}{}
% with a frontmatter-en.tex and a frontmatter-sr.tex beside this file.
\ifdefined\MaughamBody\else\newenvironment{MaughamBody}[1]{\clearpage}{}\fi

% --- Cross-body slugline links -----------------------------------------------
% In a multi-language compile every slugline links to the same slugline in each
% of the other bodies, so a reader can jump from a scene to its translation.
% Maugham emits \MaughamCrossLink{p-<tag>-<id>}{\scene{...}} — the link WRAPS
% the mode command — against \hypertarget{p-<tag>-<id>}{} targets it writes
% before every anchored node. \providecommand, so a template that defines its
% own wins; the emitted body carries a degrade-to-plain-text fallback for a
% preamble (like a pre-P3 project's) that has never heard of either.
\providecommand{\MaughamCrossLink}[2]{\hyperlink{#1}{#2}}

% Wiki link in prose: rendered as plain bold display text (no hyperlink
% target; wiki targets resolve inside Maugham, not in the PDF).
\newcommand{\wikilink}[2]{\textbf{#2}}
