# Tasks — Issue-Tracker Layer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> **Naming note (added post-Task-1):** The public Swift type is `WriterTask`, not `Task`. Swift's `_Concurrency.Task` is referenced ubiquitously in this codebase; a top-level `Task` would shadow it module-wide. All references below to the bare `Task` type — `public struct Task`, `[Task]`, `-> Task`, `Task(...)` constructors — must be read as `WriterTask`. Compound names (`TaskKind`, `TaskStatus`, `TaskAnchor`, `TaskFilter`, `TaskDeriver`, `TaskRow`, `TasksPane`, etc.) and Op-related identifiers (`.taskCreate`, `taskId`, `taskBody`, etc.) are unchanged.

**Goal:** Ship the unified task surface combining inline `- [ ]` markdown checkboxes, Fountain `[[todo:]]` boneyards, and pane-created tasks into one right-pane segment (⌘⌥5) with Document/Project scope, drag-and-drop priority reorder (op-log driven, rewindable), in-pane status toggle that rewrites the underlying text for inline tasks, single-level parent/child nesting, and MCP read-only access (`list_tasks` / `get_task`).

**Architecture:** Mirrors the annotation derivation pattern (`AnnotationDeriver` + `Document._annotationsCache` + version-token cache invalidation). Six new `OpKind` cases for pane-created task lifecycle and priority. Inline-task status changes ride existing `.typingBurst` ops. Project-scope pane-created tasks live in `.maugham/ops/__project__.jsonl` (synthetic doc id, zero new sidecar enum cases). New right-pane segment via `DetailSegment.tasks` + `DetailPaneToggle` (ADR 0005 mode-swap pattern). Clickable checkboxes in the editor route through AppKit's `textView(_:clickedOnLink:at:)` delegate — same path as existing wiki-link clicks.

**Tech stack:** Swift, SwiftUI, AppKit, XCTest. macOS 14+. xcodegen via `./gen.sh`.

**Reference spec:** [`docs/superpowers/specs/2026-05-23-tasks-design.md`](../specs/2026-05-23-tasks-design.md).

import Foundation

public enum TaskKind: String, Codable, Sendable, Equatable {
    case inlineMarkdown   = "inline_markdown"
    case fountainBoneyard = "fountain_boneyard"
    case paneCreated      = "pane_created"
}

public enum TaskStatus: String, Codable, Sendable, Equatable, CaseIterable {
    case open
    case done
    case archived
}

public struct TaskAnchor: Codable, Sendable, Equatable {
    public let docId: String
    public let paragraphId: String?

    public init(docId: String, paragraphId: String?) {
        self.docId = docId
        self.paragraphId = paragraphId
    }
}

public struct Task: Identifiable, Sendable, Equatable {
    public let id: String
    public let kind: TaskKind
    public let anchor: TaskAnchor?
    public let body: String
    public let status: TaskStatus
    public let priority: Double
    public let parentTaskId: String?
    public let createdAt: Date
    public let createdBySession: String?

    public init(
        id: String, kind: TaskKind, anchor: TaskAnchor?, body: String,
        status: TaskStatus, priority: Double, parentTaskId: String?,
        createdAt: Date, createdBySession: String?
    ) {
        self.id = id
        self.kind = kind
        self.anchor = anchor
        self.body = body
        self.status = status
        self.priority = priority
        self.parentTaskId = parentTaskId
        self.createdAt = createdAt
        self.createdBySession = createdBySession
    }
}

public struct TaskFilter: Sendable, Equatable {
    public enum Scope: Sendable, Equatable {
        case document(docId: String)
        case project
    }
    public var scope: Scope
    public var statuses: Set<TaskStatus>

    public init(scope: Scope, statuses: Set<TaskStatus> = [.open]) {
        self.scope = scope
        self.statuses = statuses
    }
}

```swift
import Foundation

public enum TaskDeriver {

    /// Pure derivation. Same inputs → same output.
    /// `docId` is the doc whose paragraphs are passed; pass `"__project__"`
    /// when deriving the project-scope op log alone.
    public static func derive(
        ops: [Op],
        paragraphs: [String: String],
        docId: String
    ) -> (tasks: [Task], rebalanceOps: [Op]) {
        // 1. Walk ops once. Pane-created records + lifecycle overrides.
        var panes: [String: PaneSeed] = [:]
        var overrides: [String: InlineOverride] = [:]
        for op in ops {
            switch op.kind {
            case .taskCreate:
                guard let id = op.provenance?.taskId,
                      let body = op.provenance?.taskBody,
                      let prio = op.provenance?.taskPriority else { continue }
                panes[id] = PaneSeed(
                    id: id, body: body, priority: prio,
                    parentTaskId: op.provenance?.taskParentId,
                    status: .open,
                    archived: false,
                    createdAt: op.at,
                    createdBySession: op.provenance?.sessionId)
            case .taskStatusChange:
                applyLifecycle(op, to: &panes, &overrides) { seed, status in
                    seed.status = TaskStatus(rawValue: status) ?? seed.status
                }
            case .taskPriorityChange:
                guard let id = op.provenance?.taskId,
                      let prio = op.provenance?.taskPriority else { continue }
                if panes[id] != nil { panes[id]?.priority = prio }
                else { overrides[id, default: .init()].priority = prio }
            case .taskParentChange:
                guard let id = op.provenance?.taskId else { continue }
                let parent = op.provenance?.taskParentId
                if panes[id] != nil { panes[id]?.parentTaskId = (parent == "" ? nil : parent) }
                else { overrides[id, default: .init()].parentTaskId = (parent == "" ? nil : parent) }
            case .taskBodyEdit:
                guard let id = op.provenance?.taskId,
                      let body = op.provenance?.taskBody else { continue }
                if panes[id] != nil { panes[id]?.body = body }
            case .taskArchive:
                guard let id = op.provenance?.taskId else { continue }
                if panes[id] != nil { panes[id]?.status = .archived }
                else { overrides[id, default: .init()].archived = true }
            default: break
            }
        }

        // 2. Line-scan paragraphs for inline `- [ ]` markdown checkboxes.
        var inlines: [Task] = []
        var seenIds: Set<String> = []
        for (paraId, text) in paragraphs {
            let lines = text.split(separator: "\n", omittingEmptySubsequences: false)
            for (lineIndex, line) in lines.enumerated() {
                if let match = MarkdownCheckboxScanner.match(String(line)) {
                    let synthId = "inline:\(docId):\(paraId):\(TaskDeriver.bodyHash(match.body))"
                    if seenIds.contains(synthId) { continue }   // collision dedupe per spec §3.2
                    seenIds.insert(synthId)
                    var task = Task(
                        id: synthId,
                        kind: .inlineMarkdown,
                        anchor: TaskAnchor(docId: docId, paragraphId: paraId),
                        body: match.body,
                        status: match.checked ? .done : .open,
                        priority: defaultTailPriority(docId: docId, line: lineIndex),
                        parentTaskId: nil,
                        createdAt: Date.distantPast,
                        createdBySession: nil)
                    task = applyOverrides(task, overrides[synthId])
                    inlines.append(task)
                }
                // (Fountain scanner — same shape, key by bodyHash; same dedupe.)
            }
        }

        // 3. Merge.
        var all: [Task] = inlines
        for seed in panes.values {
            all.append(Task(/* from PaneSeed */))
        }

        // 4. Rebalance precision check.
        let (balanced, rebalanceOps) = rebalanceIfNeeded(all)

        // 5. Sort with parent-then-child interleave.
        return (sortParentInterleaved(balanced), rebalanceOps)
    }

    // Helpers: PaneSeed, InlineOverride, applyLifecycle, applyOverrides,
    // defaultTailPriority, rebalanceIfNeeded, sortParentInterleaved, bodyHash.

    /// Normalized SHA-256 prefix per spec §3.2.
    /// Trim → lowercase → collapse internal whitespace → SHA-256 → first 8 hex chars.
    /// Exposed `internal` (not `private`) so synthetic-id tests can pin it without
    /// going through the full derive path.
    internal static func bodyHash(_ body: String) -> String {
        let normalized = body
            .trimmingCharacters(in: .whitespacesAndNewlines)
            .lowercased()
            .components(separatedBy: .whitespacesAndNewlines)
            .filter { !$0.isEmpty }
            .joined(separator: " ")
        let digest = SHA256.hash(data: Data(normalized.utf8))
        return digest.prefix(4).map { String(format: "%02x", $0) }.joined()
    }
}
```

import Foundation

extension ProjectStore {

    /// Reserved synthetic doc id for project-scope pane-created tasks.
    /// Ops written to `.maugham/ops/__project__.jsonl` are routed through
    /// the existing op-log infrastructure (autosave, presenter, rewind).
    public static let projectTasksDocId = "__project__"

    @discardableResult
    public func createProjectPaneTask(body: String, parentTaskId: String? = nil) -> Task {
        let opId = mintProjectOpId()
        let priority = lowestProjectTaskPriority() + 1.0
        let op = Op(
            opId: opId, at: Date(), kind: .taskCreate, changes: [],
            provenance: Op.Provenance(
                sessionId: currentSessionId,
                taskId: opId, taskBody: body,
                taskPriority: priority, taskParentId: parentTaskId,
                taskKind: "pane_created"))
        appendProjectTaskOp(op)
        return Task(
            id: opId, kind: .paneCreated,
            anchor: TaskAnchor(docId: Self.projectTasksDocId, paragraphId: nil),
            body: body, status: .open, priority: priority,
            parentTaskId: parentTaskId, createdAt: op.at,
            createdBySession: currentSessionId)
    }

    public func appendProjectTaskOp(_ op: Op) {
        // Append to `.maugham/ops/__project__.jsonl` using OpLogStore.
        _projectLogVersion &+= 1
        projectTasksVersion &+= 1   // SwiftUI-observable bump
    }

    public func projectTasksOpLog() -> [Op] {
        // Read `.maugham/ops/__project__.jsonl`. Empty array on first read.
    }

    // MARK: - Cross-project aggregation cache (spec §9.5)

    private struct ProjectTasksCacheKey: Equatable {
        let perDocVersionSum: Int
        let projectLogVersion: Int
    }
    private var _projectTasksCache: [Task] = []
    private var _projectTasksCacheKey: ProjectTasksCacheKey? = nil
    private var _projectLogVersion: Int = 0
    public private(set) var projectTasksVersion: Int = 0
    #if DEBUG
    internal var _debugTasksRebuildCount: Int = 0
    #endif

    private func currentCacheKey() -> ProjectTasksCacheKey {
        // Per spec §9.5: sum open doc.tasksVersion (with overflow-safe &+);
        // closed docs use mtime-hash of their op-log file as a stand-in version.
        let openSum = documentStore.openDocuments.reduce(0) { $0 &+ $1.tasksVersion }
        let closedSum = documentStore.closedDocOpLogMtimeHashSum()
        return .init(perDocVersionSum: openSum &+ closedSum,
                     projectLogVersion: _projectLogVersion)
    }

    public func listTasksAcrossProject(filter: TaskFilter) -> [Task] {
        let key = currentCacheKey()
        if _projectTasksCacheKey == key {
            return _projectTasksCache.filter(matches(filter))
        }
        // Rebuild.
        #if DEBUG
        _debugTasksRebuildCount &+= 1
        #endif
        var all: [Task] = []
        // 1. Per-doc tasks (open docs use doc.tasks; closed docs derive on the fly).
        for doc in documentStore.allRegisteredDocuments() {
            all.append(contentsOf: doc.tasksForProjectAggregation())
        }
        // 2. Project-scope pane-created tasks.
        let (projectTasks, _) = TaskDeriver.derive(
            ops: projectTasksOpLog(),
            paragraphs: [:],
            docId: Self.projectTasksDocId)
        all.append(contentsOf: projectTasks)
        _projectTasksCache = all
        _projectTasksCacheKey = key
        projectTasksVersion &+= 1
        return all.filter(matches(filter))
    }

    private func matches(_ filter: TaskFilter) -> (Task) -> Bool {
        return { task in
            guard filter.statuses.contains(task.status) else { return false }
            switch filter.scope {
            case .document(let docId): return task.anchor?.docId == docId
            case .project: return true
            }
        }
    }
}
```

- `documentStore.openDocuments` / `documentStore.allRegisteredDocuments()` / `documentStore.closedDocOpLogMtimeHashSum()` may need to be added to `DocumentStore` — verify at implementation time; pick the names already in use if equivalents exist.
- `doc.tasksForProjectAggregation()` is a thin wrapper that returns `doc.tasks(filter: TaskFilter(scope: .document(docId: doc.docId), statuses: Set(TaskStatus.allCases)))` so cached array carries every status; the final per-call status filter narrows.

import Foundation

public enum MarkdownCheckboxScanner {
    public struct Match {
        public let checked: Bool
        public let body: String
        /// Range of the `[ ]` or `[x]` glyph within the input line (UTF-16).
        public let bracketRange: Range<Int>
    }

    private static let regex = try! NSRegularExpression(
        pattern: #"^(\s*)- \[( |x)\] (.*)$"#)

    public static func match(_ line: String) -> Match? {
        let ns = line as NSString
        guard let m = regex.firstMatch(in: line, range: NSRange(location: 0, length: ns.length))
        else { return nil }
        let bracket = m.range(at: 2)
        let body = m.range(at: 3)
        let checked = ns.substring(with: bracket) == "x"
        return Match(
            checked: checked,
            body: ns.substring(with: body),
            bracketRange: (bracket.location - 1)..<(bracket.location + 2))
        // bracketRange covers the full "[ ]" or "[x]" glyph (3 chars).
    }
}
```
